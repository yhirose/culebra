33# Let there be Autograd to recursively apply the chain rule through a computation graph
34class Value {
35 new(data, children = [], local_grads = []) {
36 this.data = data # scalar value of this node calculated during forward pass
37 this.grad = 0.0 # derivative of the loss w.r.t. this node, calculated in backward pass
38 this._children = children # children of this node in the computation graph
39 this._local_grads = local_grads # local derivative of this node w.r.t. its children
40 this._visited = false
41 }
42 __add__(other) {
43 match other {
44 o: Object => Value.new(this.data + o.data, [this, o], [1.0, 1.0]),
45 _ => Value.new(this.data + other, [this], [1.0])
46 }
47 }
48 __mul__(other) {
49 match other {
50 o: Object => Value.new(this.data * o.data, [this, o], [o.data, this.data]),
51 _ => Value.new(this.data * other, [this], [other])
52 }
53 }
54 __pow__(n) { Value.new(this.data ** n, [this], [n * this.data ** (n - 1)]) }
55 __neg__() { this * (-1.0) }
56 __sub__(other) { this + (-other) }
57 __div__(other) {
58 match other {
59 o: Object => this * (o ** -1.0),
60 _ => this * (1.0 / other)
61 }
62 }
63 log() { Value.new(Math.log(this.data), [this], [1.0 / this.data]) }
64 exp() { d = Math.exp(this.data); Value.new(d, [this], [d]) }
65 relu() {
66 d = if this.data > 0.0 { this.data } else { 0.0 }
67 g = if this.data > 0.0 { 1.0 } else { 0.0 }
68 Value.new(d, [this], [g])
69 }
70 backward() {
71 mut topo = []
72 build_topo = |node| {
73 if !node._visited {
74 node._visited = true
75 for ch in node._children { self(ch) }
76 topo.push(node)
77 }
78 }
79 build_topo(this)
80 this.grad = 1.0
81 topo.reverse()
82 topo.for_each(|v| {
83 range(v._children.size()).for_each(|j| {
84 let child = v._children[j]
85 child.grad = child.grad + v._local_grads[j] * v.grad
86 })
87 })
88 for tv in topo { tv._visited = false }
89 }
90}

58 this.grad = Tensor.zeros(data.shape())
59 this._children = children
60 this._backward_fn = nil
61 this._visited = false
62 }
63 set_backward(f) { this._backward_fn = f }
64 zero_grad() { this.grad = Tensor.zeros(this.data.shape()) }
65 backward() {
66 mut topo = []
67 let build_topo = |node| {
68 if !node._visited {
69 node._visited = true
70 for ch in node._children { self(ch) }
71 topo.push(node)
72 }
73 }
74 build_topo(this)

76 Tensor.eval(this.grad)
77 topo.reverse()
78 for tv in topo {
79 if tv._backward_fn != nil { tv._backward_fn() }
80 }

82 }

170# rmsnorm of a column vector x [D, 1]: y = x * scale where
171# scale = (mean(x*x) + eps)^-0.5.
172let ts_rmsnorm = fn (x) {
173 let D = to_float(x.data.shape()[0])
174 let eps = 1.0e-5
175 let xx = x.data * x.data # [D, 1]
176 let ms = xx.sum() / D # scalar Float
177 let scale = (ms + eps) ** (-0.5) # scalar Float
178 let y_data = x.data * scale # [D, 1]
179 Tensor.eval(y_data)
180 let out = TNode.new(y_data, [x])
181 # Backward: dy_i = scale * (g_i - x_i * (sum_j g_j * x_j) * scale^2 / D)
182 # where g_i = out.grad[i].
183 out.set_backward(fn() {
184 let g = out.grad
185 let dot_gx = (g * x.data).sum() # scalar
186 let coeff = dot_gx * (scale * scale) / D # scalar
187 # dx_i = scale * (g_i - x_i * coeff)
188 x.grad += (g - x.data * coeff) * scale
189 })
190 out
191}

299# Single-head attention for the current position. With n_head=1,
300# head_dim=n_embd: q, k, v are full vectors [n_embd, 1]. KV lists
301# accumulate over positions. Custom-backward to keep the autograd
302# graph compact (avoids one TNode per per-position dot product).
303let ts_single_head_attention = fn (q, k_list, v_list, scale) {
304 let n_t = k_list.size()
305 # Forward: logits[t] = q.T @ k_list[t] * scale (scalar); weights = softmax;
306 # head_out = sum_t weights[t] * v_list[t]
307 mut logits_arr = []
308 for t in 0..n_t {
309 let dp = (q.data * k_list[t].data).sum() * scale
310 logits_arr.push([dp])
311 }
312 let logits_t = Tensor.from(logits_arr)
313 let weights_t = Tensor.softmax(logits_t.reshape([n_t]))
314 Tensor.eval(weights_t)
315 let w_arr = weights_t.to_array()
316 
317 let n_embd_local = q.data.shape()[0]
318 mut acc = Tensor.zeros(n_embd_local, 1)
319 for t in 0..n_t {
320 acc += v_list[t].data * to_float(w_arr[t])
321 }
322 Tensor.eval(acc)
323 
324 mut children = [q]
325 for kt in k_list { children.push(kt) }
326 for vt in v_list { children.push(vt) }
327 let out = TNode.new(acc, children)
328 
329 out.set_backward(fn() {
330 # dL/dhead_out is in out.grad ([n_embd, 1]).
331 # Backward through head_out = sum_t w[t] * v[t]:
332 # dL/dv[t] = out.grad * w[t]
333 # dL/dw[t] = (out.grad * v[t].data).sum() (scalar)
334 mut dL_w_arr = []
335 for t in 0..n_t {
336 let w_t = to_float(w_arr[t])
337 v_list[t].grad += out.grad * w_t
338 let dw = (out.grad * v_list[t].data).sum()
339 dL_w_arr.push(dw)
340 }
341 # Softmax backward: dL/dlogits[i] = w[i] * (dL/dw[i] - sum_j dL/dw[j]*w[j])
342 mut dot_dw_w = 0.0
343 for t in 0..n_t {
344 dot_dw_w = dot_dw_w + dL_w_arr[t] * to_float(w_arr[t])
345 }
346 mut dL_logits_arr = []
347 for t in 0..n_t {
348 let w_t = to_float(w_arr[t])
349 dL_logits_arr.push(w_t * (dL_w_arr[t] - dot_dw_w))
350 }
351 # dL/dq += sum_t dL_logits[t] * scale * k_list[t]
352 # dL/dk[t] = dL_logits[t] * scale * q
353 for t in 0..n_t {
354 let dl = dL_logits_arr[t]
355 q.grad += k_list[t].data * (dl * scale)
356 k_list[t].grad += q.data * (dl * scale)
357 }
358 })
359 out

194 range(v_h.size()).reduce(Value.new(0.0),
195 |acc, t| acc + attn_weights[t] * v_h[t][j])
196 ).collect()
197 }).collect()
198 
199 x = add_v(linear(x_attn, layer.attn_wo), x_residual1)
200

202 let x_residual2 = x
203 x = rmsnorm(x)
204 x = linear(x, layer.mlp_fc1).map(|xi| xi.relu())
205 x = add_v(linear(x, layer.mlp_fc2), x_residual2)
206 }
207 linear(x, model.lm_head)
208}

400let gpt_forward = fn (token_id, pos_id, keys, values) {
401 let tok_emb = ts_embed(wte, token_id)
402 let pos_emb = ts_embed(wpe, pos_id)
403 mut x = ts_rmsnorm(ts_add(tok_emb, pos_emb))
404

406 let x_residual1 = x
407 x = ts_rmsnorm(x)
408 let layer = layers[li]
409 
410 let q = ts_linear(layer.attn_wq, x)
411 let k = ts_linear(layer.attn_wk, x)
412 let v = ts_linear(layer.attn_wv, x)
413 keys[li].push(k)
414 values[li].push(v)
415 
416 # Multi-head attention: slice q / keys / values per head, run
417 # single-head attention for each, concat back into [n_embd, 1].
418 mut head_outs = []
419 for h in 0..n_head {
420 let hs = h * head_dim
421 let q_h = ts_slice(q, hs, hs + head_dim)
422 mut k_h_list = []
423 mut v_h_list = []
424 for t in 0..keys[li].size() {
425 k_h_list.push(ts_slice(keys[li][t], hs, hs + head_dim))
426 v_h_list.push(ts_slice(values[li][t], hs, hs + head_dim))
427 }
428 head_outs.push(ts_single_head_attention(q_h, k_h_list, v_h_list, attn_scale))
429 }
430 let x_attn = ts_concat(head_outs)
431 x = ts_add(ts_linear(layer.attn_wo, x_attn), x_residual1)
432 
433 let x_residual2 = x
434 x = ts_rmsnorm(x)
435 let h1 = ts_relu(ts_linear(layer.mlp_fc1, x))
436 x = ts_add(ts_linear(layer.mlp_fc2, h1), x_residual2)
437 }

442# === Adam optimizer ===
443#
444# Per-parameter first/second-moment buffers, same hyperparameters as
445# the scalar version.
446 
447let lr_base = 0.01
448let beta1 = 0.85
449let beta2 = 0.99
450let eps_adam = 1.0e-8
451 
452mut m_buf = []
453mut v_buf = []
454for p in params {
455 m_buf.push(Tensor.zeros(p.data.shape()))
456 v_buf.push(Tensor.zeros(p.data.shape()))
457}
458 
459let adam_step = fn (step, num_steps) {
460 let lr_t = lr_base * (1.0 - to_float(step) / to_float(num_steps))
461 let bc1 = 1.0 - beta1 ** to_float(step + 1)
462 let bc2 = 1.0 - beta2 ** to_float(step + 1)
463 for i in 0..params.size() {
464 let p = params[i]
465 let g = p.grad
466 m_buf[i] = m_buf[i] * beta1 + g * (1.0 - beta1)
467 v_buf[i] = v_buf[i] * beta2 + g * g * (1.0 - beta2)
468 Tensor.eval(m_buf[i], v_buf[i])
469 # Adam update (manually elementwise: numerator / sqrt(v_hat) is
470 # tricky without elementwise sqrt — approximate via pow(0.5).)
471 let m_hat = m_buf[i] * (1.0 / bc1)
472 let v_hat_pow = (v_buf[i] * (1.0 / bc2)).pow(0.5)
473 let denom = v_hat_pow + eps_adam
474 p.data -= (m_hat / denom) * lr_t
475 Tensor.eval(p.data)
476 }
477}

232

234 mut keys = []
235 mut values = []
236 for _ in 0..n_layer { keys.push([]); values.push([]) }
237 
238 let total_loss = range(n).reduce(Value.new(0.0), |acc, pos_id| {
239 let probs = softmax(gpt(tokens[pos_id], pos_id, keys, values))
240 acc + (-probs[tokens[pos_id + 1]].log())
241 })
242 let loss = (1.0 / n) * total_loss # final average loss over the document sequence. May yours be low.
243 
244 # Backward the loss, calculating the gradients with respect to all model parameters
245 loss.backward()
246 
247 # Adam optimizer update: update the model parameters based on the corresponding gradients
248 let lr_t = learning_rate * (1.0 - to_float(step) / to_float(num_steps)) # linear learning rate decay
249 for i in 0..params.size() {
250 let p = params[i]
251 m_buf[i] = beta1 * m_buf[i] + (1.0 - beta1) * p.grad
252 v_buf[i] = beta2 * v_buf[i] + (1.0 - beta2) * p.grad * p.grad
253 let m_hat = m_buf[i] / (1.0 - beta1 ** (step + 1))
254 let v_hat = v_buf[i] / (1.0 - beta2 ** (step + 1))
255 p.data = p.data - lr_t * m_hat / (v_hat ** 0.5 + eps_adam)
256 p.grad = 0.0
257 }

260 puts("step {step+1} / {num_steps} | loss {loss.data}")
261 }

264}
