# Forms of tools/checks/lowering_diff.py that are known to disagree and not
# yet fixed: `<sweep> <cases that disagree> <form>`, under why. A form whose
# count moves, or that agrees now, fails the gate, so a fix removes its line.
# The cause of all of them: a lowering moves a body's locals onto its state
# object by name.

binding 5 a = 1; a += 2; EMIT(a)
binding 5 a = 1; { a = 2 }; EMIT(a)
binding 2 a = 1; if C { a = 2 }; EMIT(a)
binding 5 a = 1; f = fn () { a = 2 }; f(); EMIT(a)
binding 5 z = 5; EMIT(z)
binding 1 z += 5; EMIT(z)
binding 5 println = 1; EMIT(1)
binding 5 (a, b) = (1, 2); a = 3; EMIT(a)
binding 5 (a, b) = (1, 2); EMIT(a); (a, b) = (3, 4); EMIT(a)
binding 5 let (a, b) = (1, 2); a = 3; EMIT(a)
binding 5 mut a = 1; b = 2; EMIT(a); (a, b) = (3, 4); EMIT(a)
