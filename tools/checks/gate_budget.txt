# What the gate sweeps, and how much of it. tools/checks/check_gate_budget.sh
# holds each number exactly; growing one is fine and is done here, in the commit
# that grows it, with the reason in the message.
#
# Why these and not seconds: wall-clock on a loaded machine varies by 1.5-2x
# between runs of the same gate, so it cannot be a gate. The population is
# deterministic, and it is the half that actually moved — see the per-phase
# costs in the justfile's gate table for what one item is worth on each lane
# (one tests/*.cul file: 0.06 CPU seconds under --vm, 2.2 under --jit).

# The corpus every engine lane sweeps. The expensive one: a full --jit sweep is
# 610 CPU seconds, an AOT sweep more, and three lanes of the full gate sweep it.
# 281 since test_size_too_large.cul (a size no machine can satisfy is a
# ValueError) and test_tensor_ops_window.cul (the im2col window ops' duals).
tests.cul 281

# tests/isolate/*.cul — run on both engines, plus the over-cap modes.
tests.isolate 20

# The curated bytecode cases held to frozen expected outputs, x2 lanes x3 GC
# axes in the full gate.
vm_cases 181

# ctest entries. The shell-driven ones also make up the landing gate's CLI
# phase (tools/checks/ctest_shell_driven.py names them).
ctest.entries 34

# ```culebra blocks under docs/ — `just doctest` runs the runnable ones on both
# engines, and the JIT lane costs one LLVM module per block.
# 946 since §21 shows what `@value` changes instead of asserting it: the loop
# that builds no instance, and the alias write an ordinary class accepts beside
# the one a value class refuses — two blocks, in en and ja.
docs.blocks 946

# The samples the language front ends are run over (examples/languages/*).
languages.samples 130

# The suites under examples/, swept by run_examples_sweep. They were in no gate
# until the sweep was added; the count is here so that stays visible.
examples.suites 30
