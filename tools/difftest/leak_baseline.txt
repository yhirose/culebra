# Lines starting with # are notes; blank lines are ignored.

# The gate measures under GC_NEVER, which cannot tell a reference cycle from
# a retain that is never dropped: both read as growth. The one diagnostic
# that separates them is cutting a single edge in the program (nil the
# field, reassign the binding): a cycle's growth falls to 0, a leaked
# retain's does not. Classify a new entry that way before filing it below.
#
# --- true cycles: the cycle collector's job, permanent under GC_NEVER ---
# A reference cycle the program itself writes is correct behavior with the
# collector off, not an RC bug (the same call leak_abort_suite makes for
# benign cycles). These entries stay; a "no longer leaks" report for one
# means the corpus or the collector changed, which is worth looking at
# either way.

# A closure stored on the instance whose `self` it captures is `a.me = a`
# in another spelling (instance -> closure -> self cell -> instance): the
# program wrote the cycle, and a weak `self` would change what `cb()` sees.
jit	selflex|class K4 { new(){ self.cb = fn () { type_of(self) } } }; let c = K4.new().cb; c()
vm	selflex|class K4 { new(){ self.cb = fn () { type_of(self) } } }; let c = K4.new().cb; c()
# A reassignable self-recursive binding needs its cell: the closure holds
# the cell to read the binding's current value, and the cell holds the
# closure. (The `let` twin reads itself instead — FuncInfo::own_name — so
# it no longer appears here.)
jit	nestedfn|{ mut rec = fn (n) { n > 0 ? rec(n - 1) : 20 }; rec(2) }
vm	nestedfn|{ mut rec = fn (n) { n > 0 ? rec(n - 1) : 20 }; rec(2) }
