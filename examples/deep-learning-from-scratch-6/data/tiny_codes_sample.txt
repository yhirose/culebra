def get_usage():

        cpu_usage = psutil.cpu_percent()

        gpu_temp = gpus[0].temperature

import random
def markov_chain(text, n=2):
    chain, words = {}, text.split()
    for i in range(len(words) - n):
        key, next_word = tuple(words[i:i + n - 1]), words[i + n - 1]
        chain[key] = chain.get(key, []) + [next_word]

import numpy as np
def gaussian_elimination(A, b):
    n = len(b)
    for i in range(n):
        max_row = max(range(i, n), key=lambda x: abs(A[x][i]))
        A[[i, max_row]], b[i], b[max_row] = A[[max_row, i]], b[max_row], b[i]
        for j in range(i+1, n):
            factor = A[j][i] / A[i][i]
            A[j] -= factor * A[i]
            b[j] -= factor * b[i]
    return b

def backpropagation(layers, weights, biases, activation, x, y):
    activations = [x]
    for w, b in zip(weights, biases):
        x = activation(np.dot(w, x) + b)
        activations.append(x)

import numpy as np
def FFT(x):
    N = x.shape[0]
    if N <= 1: return x
    even = FFT(x[::2])
    odd = FFT(x[1::2])
    T = np.exp(-2j * np.pi * np.arange(N // 2) / N)
    return np.concatenate([even + T * odd, even - T * odd])

def runge_kutta_4(f, x0, y0, h, n):
    x, y = x0, y0
    for i in range(n):
        k1 = h * f(x, y)
        k2 = h * f(x + 0.5 * h, y + 0.5 * k1)
        k3 = h * f(x + 0.5 * h, y + 0.5 * k2)
        k4 = h * f(x + h, y + k3)
        y += (k1 + 2 * k2 + 2 * k3 + k4) / 6
        x += h

import numpy as np
def lagrange_interpolation(x_values, y_values, x):
    result = 0
    for i in range(len(y_values)):
        term = y_values[i]
        for j in range(len(x_values)):
            if j != i:
                term = term * (x - x_values[j]) / (x_values[i] - x_values[j])
        result += term
    return result

import numpy as np
def jacobi(A, b, tol=1e-5):
    x = np.zeros_like(b)
    while True:
        x_new = np.dot(np.linalg.inv(np.diag(np.diag(A))), b - np.dot(A - np.diag(np.diag(A)), x))
        if np.allclose(x, x_new, tol):
            return x_new
        x = x_new

import numpy as np
def gauss_seidel(A, b, tol=1e-5):
    x = np.zeros_like(b)
    while True:
        x_new = np.zeros_like(x)
        for i in range(A.shape[0]):
            s1 = np.dot(A[i, :i], x[:i])
            s2 = np.dot(A[i, i + 1:], x[i + 1:])
            x_new[i] = (b[i] - s1 - s2) / A[i, i]
        if np.allclose(x, x_new, atol=tol):
            return x_new
        x = x_new

def bisection(f, a, b, tol=1e-5):
    c = (a + b) / 2.0
    while abs(f(c)) > tol:
        if f(c) * f(a) < 0:
            b = c
        else:
            a = c
        c = (a + b) / 2.0
    return c

import heapq
def dijkstra(graph, start):
    distances = {node: float('inf') for node in graph}
    distances[start] = 0
    pq = [(0, start)]
    while pq:
        current_distance, current_node = heapq.heappop(pq)
        if current_distance > distances[current_node]:
            continue
        for neighbor, weight in graph[current_node].items():
            distance = current_distance + weight
            if distance < distances[neighbor]:
                distances[neighbor] = distance
                heapq.heappush(pq, (distance, neighbor))
    return distances

def list_dict(list_one, list_two):
    """Create a dictionary of elements in the two lists which have the same index."""
    assert len(list_one) == len(list_two), "lists must be same length"
    dict = {}
    for i in range(len(list_one)):
        dict[list_one[i]] = list_two[i]
    return dict

import time
def check_fps():
    t1, cnt, intv = time.time(), 0, 1
    nxt = t1 + intv
    while True:
        cnt += 1
        time.sleep(0.001)
        t2 = time.time()
        if t2 >= nxt:
            print(f'FPS: {cnt / intv}')
            cnt, t1 = 0, t2
            nxt = t2 + intv
if __name__ == '__main__':
    check_fps()
